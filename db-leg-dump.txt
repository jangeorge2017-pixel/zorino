  66: /**
  67:  * Realistic NON-UNIFORM catalog inventory (eBay-dominant, like the production
  68:  * distribution). Providers hold their TRUE share of the matching universe —
  69:  * there is NO equal 25% allocation and NO per-provider quota. The few rare
  70:  * providers (Admitad=2, AliExpress=4) must stay fully reachable through
  71:  * pagination regardless of eBay's dominance.
  72:  *
  73:  *   eBay 388 + CJdropshipping 23 + AliExpress 4 + Admitad 2 = 417 candidates.
  74:  *
  75:  * Seed picks the provider order deterministically (mulberry32 — identical
  76:  * across platforms and Node versions), so the test never depends on runtime
  77:  * randomness and still reproduces the exact same deep-page drift every run.
  78:  */
  79: function mulberry32(seed: number): () => number {
  80:   let a = seed >>> 0;
  81:   return () => {
  82:     a |= 0;
  83:     a = (a + 0x6d2b79f5) | 0;
  84:     let t = Math.imul(a ^ (a >>> 15), 1 | a);
  85:     t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  86:     return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  87:   };
  88: }
  89: 
  90: function seededShuffle<T>(items: T[], seed: number): T[] {
  91:   const rnd = mulberry32(seed);
  92:   const out = [...items];
  93:   for (let i = out.length - 1; i > 0; i--) {
  94:     const j = Math.floor(rnd() * (i + 1));
  95:     [out[i], out[j]] = [out[j], out[i]];
  96:   }
  97:   return out;
  98: }

 100: /** 417 deterministic rows whose provider mix is intentionally NON-UNIFORM. */
 101: function makeNonUniformDbRows(count = 417): DbRow[] {
 102:   const weights: Array<[string, number]> = [
 103:     ["ebay", 388],
 104:     ["cjdropshipping", 23],
 105:     ["aliexpress", 4],
 106:     ["admitad", 2],
 107:   ];
 108:   const totalWeight = weights.reduce((s, [, w]) => s + w, 0);
 109:   if (totalWeight !== count) {
 110:     throw new Error(`non-uniform weights must sum to ${count}`);
 111:   }
 112:   const providerPool: string[] = [];
 113:   for (const [provider, w] of weights) {
 114:     for (let k = 0; k < w; k++) providerPool.push(provider);
 115:   }
 116:   const providers = seededShuffle(providerPool, 0x5eed_417);
 117:   const rows = makeDbRows(count);
 118:   return rows.map((row, i) => ({ ...row, provider: providers[i] }));
 119: }
 120: 
 121:   return [...rows].sort(
 122:     (a, b) =>
 123:       b.discount_percent - a.discount_percent ||
 124:       a.product_name.localeCompare(b.product_name) ||
 125:       a.product_id.localeCompare(b.product_id),
 126:   );
 127: }

 129: /**
 130:  * Fake supabase anon client:
 131:  *  - count leg (head): `.from("lowest_prices_today").select(_,{head:true})
 132:  *    .or().eq().eq().not().neq()` -> thenable `{ count, error }`.
 133:  *  - paged leg: `.from("lowest_prices_today").select().or().eq().eq().not()
 134:  *    .neq().order().order().order().range(from,to)` -> Promise
 135:  *    `{ data, count, error }` with count = exact total (not window-capped).
 136:  *  - products leg: `.from("products").select().in("id", ids)` -> Promise
 137:  *    `{ data, error }`.
 138:  */
 139: function buildFakeSupabase(rows: DbRow[]): () => unknown {
 140:   const ordered = serverOrdered(rows);
 141:   const total = rows.length;
 142: 
 143:   const dbChain = {
 144:     select: () => dbChain,
 145:     or: () => dbChain,
 146:     eq: () => dbChain,
 147:     not: () => dbChain,
 148:     neq: () => dbChain,
 149:     order: () => dbChain,
 150:     range: (from: number, to: number) =>
 151:       Promise.resolve({
 152:         data: ordered.slice(from, to + 1),
 153:         count: total,
 154:         error: null,
 155:       }),
 156:     then: (resolve: (v: unknown) => unknown) =>
 157:       Promise.resolve({ count: total, error: null }).then(resolve),
 158:   };
 159: 
 160:   return () => ({
 161:     from: (table: string): unknown =>
 162:       table === "products"
 163:         ? {
 164:             select: () => ({
 165:               in: (_col: string, ids: string[]) =>
 166:                 Promise.resolve({
 167:                   data: ordered
 168:                     .filter((r) => ids.includes(r.product_id))
 169:                     .map((r) => ({ id: r.product_id, category_slug: "audio" })),
 170:                   error: null,
 171:                 }),
 172:             }),
 173:           }
 174:         : dbChain,
 175:   });
 176: }
