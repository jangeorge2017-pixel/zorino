N66=/**
N67= * Realistic NON-UNIFORM catalog inventory (eBay-dominant, like the production
N68= * distribution). Providers hold their TRUE share of the matching universe —
N69= * there is NO equal 25% allocation and NO per-provider quota. The few rare
N70= * providers (Admitad=2, AliExpress=4) must stay fully reachable through
N71= * pagination regardless of eBay's dominance.
N72= *
N73= *   eBay 388 + CJdropshipping 23 + AliExpress 4 + Admitad 2 = 417 candidates.
N74= *
N75= * Seed picks the provider order deterministically (mulberry32 — identical
N76= * across platforms and Node versions), so the test never depends on runtime
N77= * randomness and still reproduces the exact same deep-page drift every run.
N78= */
N79=function mulberry32(seed: number): () => number {
N80=  let a = seed >>> 0;
N81=  return () => {
N82=    a |= 0;
N83=    a = (a + 0x6d2b79f5) | 0;
N84=    let t = Math.imul(a ^ (a >>> 15), 1 | a);
N85=    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
N86=    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
N87=  };
N88=}
N89=
N90=function seededShuffle<T>(items: T[], seed: number): T[] {
N91=  const rnd = mulberry32(seed);
N92=  const out = [...items];
N93=  for (let i = out.length - 1; i > 0; i--) {
N94=    const j = Math.floor(rnd() * (i + 1));
N95=    [out[i], out[j]] = [out[j], out[i]];
N96=  }
N97=  return out;
N98=}

N100=/** 417 deterministic rows whose provider mix is intentionally NON-UNIFORM. */
N101=function makeNonUniformDbRows(count = 417): DbRow[] {
N102=  const weights: Array<[string, number]> = [
N103=    ["ebay", 388],
N104=    ["cjdropshipping", 23],
N105=    ["aliexpress", 4],
N106=    ["admitad", 2],
N107=  ];
N108=  const totalWeight = weights.reduce((s, [, w]) => s + w, 0);
N109=  if (totalWeight !== count) {
N110=    throw new Error(`non-uniform weights must sum to ${count}`);
N111=  }
N112=  const providerPool: string[] = [];
N113=  for (const [provider, w] of weights) {
N114=    for (let k = 0; k < w; k++) providerPool.push(provider);
N115=  }
N116=  const providers = seededShuffle(providerPool, 0x5eed_417);
N117=  const rows = makeDbRows(count);
N118=  return rows.map((row, i) => ({ ...row, provider: providers[i] }));
N119=}
N120=
N121=  return [...rows].sort(
N122=    (a, b) =>
N123=      b.discount_percent - a.discount_percent ||
N124=      a.product_name.localeCompare(b.product_name) ||
N125=      a.product_id.localeCompare(b.product_id),
N126=  );
N127=}
